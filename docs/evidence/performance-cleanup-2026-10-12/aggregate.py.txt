import json, math, statistics, pathlib, html

root = pathlib.Path('/home/kristjan/code/agent-jobs/docs')
evidence = root / 'evidence/performance-cleanup-2026-10-12'
evidence.mkdir(parents=True, exist_ok=True)
def percentile(xs, p):
    return sorted(xs)[math.ceil(len(xs)*p)-1] if xs else None
def metric(xs):
    return {'median': statistics.median(xs) if xs else None, 'p75': percentile(xs,.75)}
reports = {}
for label in ('before','after'):
    raw = pathlib.Path('/tmp/sidequest-cleanup-'+label+'.json').read_text()
    report = json.loads(raw)
    (evidence / (label+'.json')).write_text(raw)
    groups=[]
    for route in ['/jobs','/job/84','/agent/2036','/services']:
      for kind in ['cold','warm']:
        runs=[r for r in report['runs'] if r['route']==route and r['kind']==kind]
        done=[r for r in runs if isinstance(r.get('readyMs'),(int,float)) and r['readyMs']<20000 and not r.get('timedOut')]
        groups.append({'route':route,'kind':kind,'n':len(runs),'completed':len(done),
          'readyMs':metric([r['readyMs'] for r in done]),
          'cappedReadyMs':metric([r['readyMs'] if r in done else 20000 for r in runs]),
          'lcp':metric([r['lcp'] for r in runs if r.get('lcp')]),
          'cls':metric([r['cls'] for r in runs if 'cls' in r]),
          'reads':metric([len(r['requests']) for r in runs]),
          'scriptTransferBytes':metric([r['scriptTransferBytes'] for r in runs if 'scriptTransferBytes' in r]),
          'listReads':sum(1 for r in runs for q in r['requests'] if q['path']=='/data/jobs' or q['path'].endswith('/task_index')),
          'pageErrors':sum(r.get('pageErrors',0) for r in runs),
          'httpErrors':sum(r.get('httpErrors',0) for r in runs),
          'pending':sum(r.get('pending',0) for r in runs)})
    reports[label]={k:v for k,v in report.items() if k!='runs'}|{'groups':groups}
(evidence/'summary.json').write_text(json.dumps(reports,indent=2)+'\n')
